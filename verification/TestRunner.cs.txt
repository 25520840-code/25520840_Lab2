internal static class TestHooks
{
    public static string BasePath = Path.GetFullPath(Path.Combine(AppContext.BaseDirectory, "../../../evidence"));
    public static string DataPath = Path.Combine(BasePath, "appdata");
    public static Queue<(string Path, DialogResult Result)> Dialogs = new();
    public static List<string> Messages = new();
    public static DialogResult ShowDialog(FileDialog dialog)
    {
        if (Dialogs.Count == 0) throw new Exception("Missing dialog fixture");
        var next = Dialogs.Dequeue(); dialog.FileName = next.Path; return next.Result;
    }
    public static DialogResult ShowMessage(string text, params object[] args)
    { Messages.Add(text); return args.OfType<MessageBoxButtons>().Any(x=>x==MessageBoxButtons.YesNo) ? DialogResult.Yes : DialogResult.OK; }
}

    static BindingFlags Flags = BindingFlags.NonPublic | BindingFlags.Public | BindingFlags.Instance;
    static T Field<T>(object obj, string name) => (T)obj.GetType().GetField(name, Flags)!.GetValue(obj)!;
    static object? Call(object obj, string name, params object[] args)
    {
        try { return obj.GetType().GetMethod(name,Flags)!.Invoke(obj,args); }
        catch(TargetInvocationException e) { throw e.InnerException!; }
    }
    static void Click(Form f,string name) { Field<Button>(f,name).PerformClick(); Application.DoEvents(); }

    { Results.Add(new {id,task,action,expected,actual,pass}); Console.WriteLine($"{id}: {(pass?"PASS":"FAIL")} {actual}"); }

    {
        f.Refresh(); Application.DoEvents();
        using var bmp=new Bitmap(f.Width,f.Height);
        f.DrawToBitmap(bmp,new Rectangle(0,0,f.Width,f.Height));
        bmp.Save(P(name+".png"));
    }
